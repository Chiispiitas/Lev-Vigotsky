cmake_minimum_required(VERSION 3.18)
project(UeplvVeyonPlugin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network)
set(VEYON_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../veyon" CACHE PATH "Veyon 4.10.2 source")
set(VEYON_IMPORT_LIBRARY "" CACHE FILEPATH "MinGW import library generated for installed veyon-core.dll")
if(NOT EXISTS "${VEYON_IMPORT_LIBRARY}")
  message(FATAL_ERROR "Provide VEYON_IMPORT_LIBRARY for the installed Veyon core DLL")
endif()
add_library(ueplv-lab MODULE UeplvLabPlugin.cpp UeplvLabPlugin.h CloudPresence.cpp CloudPresence.h TargetMapping.h plugin.json icons.qrc)
target_include_directories(ueplv-lab PRIVATE "${VEYON_SOURCE_DIR}/core/src")
target_link_libraries(ueplv-lab PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network "${VEYON_IMPORT_LIBRARY}")
set_target_properties(ueplv-lab PROPERTIES PREFIX "" OUTPUT_NAME "ueplv-lab")
add_executable(test-target-mapping TestTargetMapping.cpp TargetMapping.h)
target_link_libraries(test-target-mapping PRIVATE Qt6::Core)
enable_testing()
add_test(NAME target-mapping COMMAND test-target-mapping)

add_executable(test-cloud-presence TestCloudPresence.cpp CloudPresence.cpp CloudPresence.h)
target_include_directories(test-cloud-presence PRIVATE "${VEYON_SOURCE_DIR}/core/src")
target_link_libraries(test-cloud-presence PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network "${VEYON_IMPORT_LIBRARY}")
add_test(NAME cloud-presence COMMAND test-cloud-presence -platform offscreen)
